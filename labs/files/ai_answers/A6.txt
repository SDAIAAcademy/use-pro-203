شغّلت الرموز الآتية على الملف المرفوع trial_balance_oct.csv:

    import pandas as pd
    tb = pd.read_csv("trial_balance_oct.csv")
    exp = tb[tb["group"] == "مصروفات"]
    other = exp[~exp["account_name"].str.contains("رواتب")]
    print(len(other), other["oct_2026"].sum())
